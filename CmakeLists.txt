cmake_minimum_required(VERSION 3.16)

project(Singleton)

add_library(${PROJECT_NAME}_single_lib) 

target_include_directories(${PROJECT_NAME}_single_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

target_sources(${PROJECT_NAME}_single_lib
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src/single_thread/single_logic.cpp
)

add_executable(${PROJECT_NAME}_single)

target_sources(${PROJECT_NAME}_single PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/single_thread/main.cpp)

target_link_libraries(${PROJECT_NAME}_single ${PROJECT_NAME}_single_lib)


add_library(${PROJECT_NAME}_multi_lib)

target_include_directories(${PROJECT_NAME}_multi_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

target_sources(${PROJECT_NAME}_multi_lib
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src/multi_thread/multi_logic.cpp
)

add_executable(${PROJECT_NAME}_multi)

target_sources(${PROJECT_NAME}_multi PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/multi_thread/main.cpp)

target_link_libraries(${PROJECT_NAME}_multi ${PROJECT_NAME}_multi_lib)
